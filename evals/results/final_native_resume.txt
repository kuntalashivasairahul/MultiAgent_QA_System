"""Operational resumption only: unchanged inputs, no fresh attempts or verdict replacement."""
import hashlib
import json
import sys
import time
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
sys.path.insert(0, str(ROOT))
import graph_v2_1 as v21
from evals.run_eval import RESULTS, load_golden, run_inputs, run_eval, _load_partial, _evaluation_lock

def recovery_delay(rows):
    details = [detail for row in rows
               for detail in [row.get("generation_error_details") or {},
                              *(j.get("error_details") or {} for j in row.get("judges", {}).values())]]
    observed = [(detail, float(detail["retry_after"])) for detail in details
                if str(detail.get("retry_after") or "").replace(".", "", 1).isdigit()]
    # shortcut: this free profile refills ~8.3k tokens/hour; reserve 2h for quote repair, revisit for other budgets.
    reserve = 7200 if any(d.get("validation_failures") and d.get("status_code") == 429 and seconds > 60
                          for d, seconds in observed) else 35
    return max([reserve, *(seconds + 5 for _, seconds in observed)])


def wait_for_recovery(partial):
    if not partial.exists():
        return
    while True:
        try:
            with _evaluation_lock(partial.parent / "owner.sqlite"):
                rows = list(_load_partial(partial).values())
                modified = partial.stat().st_mtime
            break
        except RuntimeError as exc:
            if "This evaluation is already running" not in str(exc):
                raise
            print("Existing evaluation owns this checkpoint; waiting 30 seconds.", flush=True)
            time.sleep(30)
    if not any(r.get("generation_error") or any(j.get("error") for j in r.get("judges", {}).values()) for r in rows):
        return
    delay = max(0, recovery_delay(rows) - max(0, time.time() - modified))
    if not delay:
        return
    print("Provider recovery wait seconds:", delay, "checkpoint:", str(partial), flush=True)
    deadline = time.monotonic() + delay
    while (remaining := deadline - time.monotonic()) > 0:
        time.sleep(min(remaining, 60))


assert recovery_delay([{"generation_error_details": None}]) == 35
assert recovery_delay([{"generation_error_details": {"retry_after": "120"}}]) == 125
assert recovery_delay([{"generation_error_details": {"status_code": 429, "retry_after": "1446",
                                                    "validation_failures": [{}]}}]) == 7200
assert recovery_delay([{"generation_error_details": {"status_code": 429, "retry_after": "20",
                                                    "validation_failures": [{}]}}]) == 35
assert recovery_delay([{"judges": {"correct": {"error_details": {"retry_after": "7.5"}}}}]) == 35
assert (ROOT / "evals/confirmation_prose_v21.jsonl").exists()
for dataset in (ROOT / "evals/golden_current_v21.jsonl", ROOT / "evals/confirmation_prose_v21.jsonl"):
    inputs = {**run_inputs(v21, load_golden(dataset=dataset), dataset), "limit": None, "variant": None}
    fingerprint = hashlib.sha256(json.dumps(inputs, sort_keys=True).encode()).hexdigest()
    run_root = RESULTS / "verified" / "graph_v2_1" / fingerprint
    attempts = sorted(p for p in run_root.glob("*") if p.is_dir())
    if attempts:
        wait_for_recovery(attempts[-1] / "partial.jsonl")
    identity = None
    for cycle in range(1, 65):
        while True:
            try:
                result = run_eval("graph_v2_1", dataset=dataset)
                break
            except RuntimeError as exc:
                if "This evaluation is already running" not in str(exc):
                    raise
                print("Existing evaluation owns this attempt; waiting 30 seconds.", flush=True)
                time.sleep(30)
        current = (result["fingerprint"], result["run_id"])
        if identity is None:
            identity = current
        assert current == identity, "Resumption must retain the same inputs and attempt"
        print(json.dumps({"dataset": dataset.name, "cycle": cycle, "status": result["status"],
                          "coverage": result["coverage"], "artifact_dir": result["artifact_dir"]}), flush=True)
        if result["status"] == "complete":
            break
        wait_for_recovery(Path(result["artifact_dir"]) / "partial.jsonl")
    else:
        raise SystemExit("Still incomplete after 64 resumptions; preserve artifacts and inspect operational errors")
